WRITE YOUR BOT — FULL STARTER CODE
Copy this file, add your credentials, and run it. It logs in, fetches active markets, and places a YES order on the first market that matches your filter:
PYTHON
from kalshi_python.api import KalshiApi
from kalshi_python.models import LoginRequest, CreateOrderRequest
import time

EMAIL    = "your@email.com"      # replace
PASSWORD = "your_password"       # replace

api = KalshiApi()

# Log in and get session token
login_resp = api.login(LoginRequest(email=EMAIL, password=PASSWORD))
print("Logged in:", login_resp.member_id)

# Fetch active markets
markets = api.get_markets(status="open").markets

for market in markets:
    ticker = market.ticker
    yes_price = market.yes_ask  # cents (0-99)

    # Example filter: only buy if YES price is under 30¢
    if yes_price and yes_price < 30:
        order = api.create_order(CreateOrderRequest(
            ticker  = ticker,
            side    = "yes",
            count   = 1,         # number of contracts
            type    = "market",
        ))
        print(f"Placed YES order on {ticker}: {order.order_id}")
        break

    time.sleep(0.1)  # be polite to the API
💡 Start with the demo environment first — use api = KalshiApi(host="demo-api.kalshi.co") to test without real money.

BUILD YOUR WEBHOOK RECEIVER SERVER
Run this Python Flask server — it receives TradingView alerts and forwards them to your broker's API. Deploy it on Railway or Render for free:
PYTHON
from flask import Flask, request, jsonify
import requests, os

app = Flask(__name__)

BROKER_API_KEY = os.environ.get("BROKER_API_KEY")  # set in env vars

@app.route("/webhook", methods=["POST"])
def webhook():
    data = request.get_json()
    action = data.get("action")   # "buy" or "sell"
    ticker = data.get("ticker")
    price  = data.get("price")

    print(f"Signal received: {action} {ticker} @ {price}")

    # --- PLACE ORDER WITH YOUR BROKER ---
    # Example: OANDA REST API
    headers = {"Authorization": f"Bearer {BROKER_API_KEY}", "Content-Type": "application/json"}
    order_body = {
        "order": {
            "units": "1000" if action == "buy" else "-1000",
            "instrument": ticker,
            "timeInForce": "FOK",
            "type": "MARKET",
        }
    }
    r = requests.post("https://api-fxtrade.oanda.com/v3/accounts/YOUR_ID/orders",
                      json=order_body, headers=headers)
    return jsonify({"status": "ok", "broker_response": r.status_code})

if __name__ == "__main__":
    app.run(port=5000)

PASTE YOUR EA CODE
Replace the template with this EMA cross EA — it runs automatically on every new candle:
MQL5
//+------------------------------------------------------------------+
//| EMA Cross EA — paste into MetaEditor                           |
//+------------------------------------------------------------------+
input int    FastEMA  = 9;
input int    SlowEMA  = 21;
input double LotSize  = 0.1;
input int    StopLoss = 50;   // pips
input int    TakeProfit = 100; // pips

int fastHandle, slowHandle;

int OnInit() {
    fastHandle = iMA(_Symbol, PERIOD_CURRENT, FastEMA, 0, MODE_EMA, PRICE_CLOSE);
    slowHandle = iMA(_Symbol, PERIOD_CURRENT, SlowEMA, 0, MODE_EMA, PRICE_CLOSE);
    return INIT_SUCCEEDED;
}

void OnTick() {
    double fast[], slow[];
    CopyBuffer(fastHandle, 0, 0, 3, fast);
    CopyBuffer(slowHandle, 0, 0, 3, slow);

    bool bullCross = fast[1] > slow[1] && fast[2] <= slow[2];
    bool bearCross = fast[1] < slow[1] && fast[2] >= slow[2];

    double ask = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
    double bid = SymbolInfoDouble(_Symbol, SYMBOL_BID);
    double pip = _Point * 10;

    MqlTradeRequest req = {};
    MqlTradeResult  res = {};

    if (bullCross) {
        req.action   = TRADE_ACTION_DEAL;
        req.symbol   = _Symbol;
        req.volume   = LotSize;
        req.type     = ORDER_TYPE_BUY;
        req.price    = ask;
        req.sl       = ask - StopLoss * pip;
        req.tp       = ask + TakeProfit * pip;
        req.magic    = 42;
        OrderSend(req, res);
    }

    if (bearCross) {
        req.action   = TRADE_ACTION_DEAL;
        req.symbol   = _Symbol;
        req.volume   = LotSize;
        req.type     = ORDER_TYPE_SELL;
        req.price    = bid;
        req.sl       = bid + StopLoss * pip;
        req.tp       = bid - TakeProfit * pip;
        req.magic    = 42;
        OrderSend(req, res);
    }
}

BUILD A SIMPLE TRADING LOOP
PYTHON
import robin_stocks.robinhood as rh
import time, pyotp

rh.login(USERNAME, PASSWORD, mfa_code=pyotp.TOTP(TOTP_KEY).now())

def get_ema(prices, period):
    k = 2 / (period + 1)
    ema = prices[0]
    for p in prices[1:]:
        ema = p * k + ema * (1 - k)
    return ema

SYMBOL = "TSLA"
owned  = False

while True:
    # Fetch last 10 closing prices
    historicals = rh.stocks.get_stock_historicals(SYMBOL, interval="5minute", span="day")
    closes = [float(h["close_price"]) for h in historicals[-20:]]

    fast_ema = get_ema(closes, 9)
    slow_ema = get_ema(closes, 21)

    price = float(rh.stocks.get_latest_price(SYMBOL)[0])
    print(f"{SYMBOL}: ${price:.2f} | EMA9={fast_ema:.2f} EMA21={slow_ema:.2f}")

    if fast_ema > slow_ema and not owned:
        rh.orders.order_buy_fractional_by_price(SYMBOL, 100, "gfd")
        owned = True
        print("BUY signal — order placed")

    elif fast_ema < slow_ema and owned:
        rh.orders.order_sell_fractional_by_price(SYMBOL, 100, "gfd")
        owned = False
        print("SELL signal — order placed")
